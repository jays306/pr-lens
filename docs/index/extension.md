# src/types.ts
DOES: shared data contracts between backend SSE stream and overlay UI
TYPE: RiskLevel = "low"|"medium"|"high"|"critical"
TYPE: PRCategory { id, icon, label, summary, snippets: CodeSnippet[], riskLevel, reviewQuestions, fileCount }
TYPE: CodeSnippet { file, language, before, after, lineStart, explanation, riskLevel }
TYPE: PRAnalysis { title, author, riskScore, riskLevel, summary, criticalityLabel, affectedSystems, suggestedReviewOrder, categories, finalRecommendation? }
TYPE: SSEEvent = discriminated union on type: meta|risk|summary|systems|comments|category|recommendation|done|error

# src/content.ts
DOES: Chrome content script entry — detects PR URLs on GitHub/GitLab/Bitbucket, reads backendUrl from chrome.storage, creates JustPROverlay, handles SPA nav via MutationObserver
SYMBOLS: detectPRUrl() → string|null, init(backendUrl) → void
CALLS: overlay.setBackendUrl, overlay.JustPROverlay, chrome.storage.local
CONFIG: DEFAULT_BACKEND_URL = "http://localhost:8080"

# src/overlay.ts
DOES: entire PR review panel — Shadow DOM, multi-screen state machine (idle→loading→overview→step→decision), SSE consumption, syntax highlighting, file tree, diff rendering, comment authoring, GitHub review submission
TYPE: AnalysisState { riskScore, riskLevel, riskLabel, summary, affected[], reviewOrder[], categories, existingComments, recommendation? }
TYPE: Screen = "idle"|"loading"|"overview"|"step"|"decision"
SYMBOLS: setBackendUrl(url) → void, class JustPROverlay { constructor(prUrl), startAnalysis(), destroy(), renderScreen(screen), renderOverview(), renderStep(catIdx), renderDecision(), consumeStream(response), submitReview(), highlight(code, lang), buildFileTree(paths), buildSnippet(s, catIdx, snippetIdx), buildCommentForm/Note, reset(), togglePanel(), initResize() }
CALLS: __generated_css.overlayCSS, marked.parse, hljs.highlight, fetch (POST /analyze, POST /review)
USE WHEN: any UI change — edit this file + overlay.css, then rebuild

# src/__generated_css.ts
DOES: auto-generated; exports raw CSS string for Shadow DOM injection
SYMBOLS: overlayCSS: string
USE WHEN: never hand-edit — regenerated by build.ts

# src/styles/overlay.css
DOES: complete visual styling for Shadow DOM panel — CSS custom properties, toggle, slide-in panel, loading, overview, step layout, decision screen, comment forms, progress bar
CONFIG: --panel-width: 1080px, --copper: #9a5216, toggle z-index: 2147483646, panel z-index: 2147483645

# src/testbed.ts
DOES: standalone test harness for UI dev without real GitHub page
CALLS: overlay.JustPROverlay

# public/popup.js
DOES: extension popup — loads/saves backendUrl from chrome.storage, pings /health with 2.5s timeout
SYMBOLS: checkHealth(baseUrl) → void
CONFIG: DEFAULT_BACKEND_URL = "http://localhost:8080"

# build.ts
DOES: Bun build — reads overlay.css → writes __generated_css.ts, bundles content.ts + testbed.ts to dist/
CONFIG: NODE_ENV=production enables minify

# manifest.json
DOES: Chrome MV3 manifest — content script on PR URLs, popup, host permissions for GitHub/GitLab/Bitbucket
CONFIG: permissions: activeTab, scripting, storage; run_at: document_idle
